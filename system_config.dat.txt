def clean_text(text):
    punctuation = ".,!?;:()[]{}\"'"
    text = text.lower()
    for mark in punctuation:
        text = text.replace(mark, " ")
    return text.split()


def analyse_text(text):
    words = clean_text(text)
    if len(words) == 0:
        print("Please enter some text.")
        return

    frequency = {}
    for word in words:
        if word in frequency:
            frequency[word] += 1
        else:
            frequency[word] = 1

    top_words = sorted(frequency.items(), key=lambda item: item[1], reverse=True)
    longest = max(words, key=len)
    average = sum(len(word) for word in words) / len(words)
    palindromes = []
    for word in words:
        if len(word) > 1 and word == word[::-1] and word not in palindromes:
            palindromes.append(word)

    print("Total words:", len(words))
    print("Unique words:", len(frequency))
    print("Top 5 words:", top_words[:5])
    print("Longest word:", longest)
    print("Average word length:", round(average, 2))
    print("Palindromes:", palindromes)


def main():
    paragraph = input("Enter a paragraph: ")
    analyse_text(paragraph)


if __name__ == "__main__":
    main()

def read_matrix(rows, columns):
    matrix = []
    for i in range(rows):
        row = []
        for j in range(columns):
            value = int(input("Enter element: "))
            row.append(value)
        matrix.append(row)
    return matrix


def add_matrices(A, B):
    result = []
    for i in range(len(A)):
        row = []
        for j in range(len(A[0])):
            row.append(A[i][j] + B[i][j])
        result.append(row)
    return result


def subtract_matrices(A, B):
    result = []
    for i in range(len(A)):
        row = []
        for j in range(len(A[0])):
            row.append(A[i][j] - B[i][j])
        result.append(row)
    return result


def multiply_matrices(A, B):
    result = []
    for i in range(len(A)):
        row = []
        for j in range(len(B[0])):
            total = 0
            for k in range(len(B)):
                total += A[i][k] * B[k][j]
            row.append(total)
        result.append(row)
    return result


def transpose_matrix(A):
    result = []
    for j in range(len(A[0])):
        row = []
        for i in range(len(A)):
            row.append(A[i][j])
        result.append(row)
    return result


def main():
    r1 = int(input("Rows of first matrix: "))
    c1 = int(input("Columns of first matrix: "))
    if r1 <= 0 or c1 <= 0:
        print("Matrix dimensions must be positive.")

    A = read_matrix(r1, c1)
    r2 = int(input("Rows of second matrix: "))
    c2 = int(input("Columns of second matrix: "))
    if r2 <= 0 or c2 <= 0:
        print("Matrix dimensions must be positive.")
        return
    B = read_matrix(r2, c2)

    if r1 == r2 and c1 == c2:
        print("Addition:", add_matrices(A, B))
        print("Subtraction:", subtract_matrices(A, B))
    else:
        print("Addition and subtraction are not possible.")

    if c1 == r2:
        print("Multiplication:", multiply_matrices(A, B))
    else:
        print("Multiplication is not possible.")

    print("Transpose of first matrix:", transpose_matrix(A))

def tower_of_hanoi(n, source, target, helper):
    if n == 1:
        print("Move disk from", source, "to", target)
        return 1
    moves = tower_of_hanoi(n - 1, source, helper, target)
    print("Move disk from", source, "to", target)
    moves += 1
    moves += tower_of_hanoi(n - 1, helper, target, source)
    return moves


def binary_search(numbers, low, high, key):
    if low > high:
        return -1
    middle = (low + high) // 2
    if numbers[middle] == key:
        return middle
    if key < numbers[middle]:
        return binary_search(numbers, low, middle - 1, key)
    return binary_search(numbers, middle + 1, high, key)


def permutations(word):
    if len(word) <= 1:
        return [word]
    result = []
    for i in range(len(word)):
        first = word[i]
        remaining = word[:i] + word[i + 1:]
        for part in permutations(remaining):
            value = first + part
            if value not in result:
                result.append(value)
    return result

def jaccard(user1, user2):
    common = courses[user1] & courses[user2]
    all_courses = courses[user1] | courses[user2]
    return len(common) / len(all_courses)


def recommend(user):
    if user not in courses:
        print("User not found.")
        return
    suggested = set()
    for other in courses:
        if other != user and jaccard(user, other) > 0:
            suggested = suggested | (courses[other] - courses[user])
    if len(suggested) == 0:
        print("No recommendations found.")
    else:
        print("Recommended courses:", list(suggested)[:3])


def main():
    user = input("Enter user name: ")
    if user in courses:
        action = input("Add or remove a course? (a/r/n): ").lower()
        if action == "a":
            courses[user].add(input("Course name: "))
        elif action == "r":
            courses[user].discard(input("Course name: "))
        print("Your courses:", courses[user])
        recommend(user)
    else:
        print("User not found.")

    popular = {}
    for user_courses in courses.values():
        for course in user_courses:
            popular[course] = popular.get(course, 0) + 1
    most_popular = max(popular, key=popular.get)
    print("Most popular course:", most_popular)

def process_sales():
    region_total = {}
    valid_rows = []
    try:
        file = open("sales.txt", "r")
    except FileNotFoundError:
        print("sales.txt file not found.")
        return

    for line in file:
        parts = line.strip().split(",")
        if len(parts) != 5:
            continue
        try:
            sale_id = parts[0].strip()
            region = parts[1].strip()
            product = parts[2].strip()
            price = float(parts[3])
            quantity = int(parts[4])
            if price < 0 or quantity < 0:
                continue
            revenue = price * quantity
            region_total[region] = region_total.get(region, 0) + revenue
            valid_rows.append(line.strip() + "," + str(revenue) + "\n")
        except ValueError:
            continue
    file.close()

    with open("sales_summary.txt", "w") as output:
        output.write("Valid sales records with revenue:\n")
        for row in valid_rows:
            output.write(row)
        output.write("\nRegion totals:\n")
        for region in region_total:
            output.write(region + ": " + str(region_total[region]) + "\n")
        if region_total:
            best = max(region_total, key=region_total.get)
            output.write("Highest revenue region: " + best)

    print("Region totals:", region_total)
    if region_total:
        print("Highest revenue region:", max(region_total, key=region_total.get))
    print("Summary saved in sales_summary.txt")

class InvalidTransactionError(Exception):
    pass


def process_transactions():
    balances = {}
    rejected = 0
    try:
        file = open("transactions.txt", "r")
    except FileNotFoundError:
        print("transactions.txt file not found.")
        return

    for line in file:
        parts = line.strip().split(",")
        if len(parts) != 3:
            rejected += 1
            continue
        account = parts[0].strip()
        transaction = parts[1].strip().lower()
        try:
            amount = float(parts[2])
            if amount <= 0:
                raise InvalidTransactionError("Amount must be positive")
            if account not in balances:
                balances[account] = 0
            if transaction == "deposit":
                balances[account] += amount
            elif transaction == "withdraw":
                if amount > balances[account]:
                    raise InvalidTransactionError("Insufficient balance")
                balances[account] -= amount
            else:
                raise InvalidTransactionError("Wrong transaction type")
        except (ValueError, InvalidTransactionError):
            rejected += 1
    file.close()
    print("Final balances:", balances)
    print("Rejected transactions:", rejected)


if __name__ == "__main__":
    process_transactions()

class Book:
    def __init__(self, book_id, title):
        self.book_id = book_id
        self.title = title
        self.issued = False


class Member:
    def __init__(self, member_id, name):
        self.member_id = member_id
        self.name = name
        self.books = []
        self.fine = 0


class Library:
    def __init__(self):
        self.books = {}
        self.members = {}

    def issue_book(self, member_id, book_id):
        if member_id not in self.members or book_id not in self.books:
            print("Invalid member or book.")
            return
        member = self.members[member_id]
        book = self.books[book_id]
        if book.issued:
            print("Book is already issued.")
        elif len(member.books) == 3:
            print("A member can issue only 3 books.")
        else:
            member.books.append(book_id)
            book.issued = True
            print("Book issued.")

    def return_book(self, member_id, book_id, late_days):
        if member_id not in self.members or book_id not in self.books:
            print("Invalid member or book.")
            return
        member = self.members[member_id]
        book = self.books[book_id]
        if book_id not in member.books:
            print("This book was not issued to this member.")
            return
        member.books.remove(book_id)
        book.issued = False
        if late_days > 0:
            member.fine += late_days * 2
        print("Book returned. Fine:", member.fine)

    def display(self):
        for member in self.members.values():
            book_titles = []
            for book_id in member.books:
                book_titles.append(self.books[book_id].title)
            print(member.name, "Issued books:", book_titles)
            if member.fine > 0:
                print(member.name, "Fine: Rs.", member.fine)

lib.members[101] = Member(101, "Amit")

class Wallet:
    def __init__(self, pin, balance=0):
        self.__pin = pin
        self.__balance = balance
        self.__wrong_attempts = 0
        self.__locked = False
        self.history = []

    def check_pin(self, pin):
        if self.__locked:
            print("Wallet is locked.")
            return False
        if pin == self.__pin:
            self.__wrong_attempts = 0
            return True
        self.__wrong_attempts += 1
        print("Wrong PIN.")
        if self.__wrong_attempts == 3:
            self.__locked = True
            print("Wallet locked after 3 wrong attempts.")
        return False

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            self.history.append("Deposited " + str(amount))
        else:
            print("Enter a positive amount.")

    def withdraw(self, pin, amount):
        if self.check_pin(pin):
            if amount > 0 and amount <= self.__balance:
                self.__balance -= amount
                self.history.append("Withdrawn " + str(amount))
            else:
                print("Invalid amount or low balance.")

    def change_pin(self, old_pin, new_pin):
        if self.check_pin(old_pin):
            self.__pin = new_pin
            print("PIN changed.")

    def show_statement(self):
        print("Transactions:", self.history)
        print("Balance:", self.__balance)

class Polynomial:
    def __init__(self, coefficients):
        self.coefficients = coefficients  # Constant term comes first

    def __str__(self):
        answer = ""
        for power in range(len(self.coefficients) - 1, -1, -1):
            value = self.coefficients[power]
            if value != 0:
                answer += str(value) + "x^" + str(power) + " + "
        if answer == "":
            return "0"
        return answer[:-3]

    def __add__(self, other):
        size = max(len(self.coefficients), len(other.coefficients))
        result = []
        for i in range(size):
            a = self.coefficients[i] if i < len(self.coefficients) else 0
            b = other.coefficients[i] if i < len(other.coefficients) else 0
            result.append(a + b)
        return Polynomial(result)

    def __sub__(self, other):
        size = max(len(self.coefficients), len(other.coefficients))
        result = []
        for i in range(size):
            a = self.coefficients[i] if i < len(self.coefficients) else 0
            b = other.coefficients[i] if i < len(other.coefficients) else 0
            result.append(a - b)
        return Polynomial(result)

    def __mul__(self, other):
        result = [0] * (len(self.coefficients) + len(other.coefficients) - 1)
        for i in range(len(self.coefficients)):
            for j in range(len(other.coefficients)):
                result[i + j] += self.coefficients[i] * other.coefficients[j]
        return Polynomial(result)

    def __call__(self, x):
        answer = 0
        for i in range(len(self.coefficients)):
            answer += self.coefficients[i] * (x ** i)
        return answer

    def derivative(self):
        result = []
        for i in range(1, len(self.coefficients)):
            result.append(i * self.coefficients[i])
        return Polynomial(result)

class Product:
    def __init__(self, code, name, price, quantity):
        if price < 0 or quantity < 0:
            raise ValueError("Price and quantity cannot be negative.")
        self.code = code
        self.name = name
        self.price = price
        self.quantity = quantity


class Inventory:
    def __init__(self):
        self.products = {}

    def add_product(self, product):
        self.products[product.code] = product

    def restock(self, code, quantity):
        if code in self.products and quantity > 0:
            self.products[code].quantity += quantity
        else:
            print("Invalid product or quantity.")

    def sell(self, code, quantity):
        if code not in self.products or quantity <= 0:
            print("Invalid product or quantity.")
        elif quantity > self.products[code].quantity:
            print("Not enough stock.")
        else:
            self.products[code].quantity -= quantity
            print("Sale completed.")

    def total_value(self):
        total = 0
        for product in self.products.values():
            total += product.price * product.quantity
        return total

    def low_stock(self, limit):
        for product in self.products.values():
            if product.quantity < limit:
                print(product.name, product.quantity)

    def save_file(self):
        with open("inventory.txt", "w") as file:
            for product in self.products.values():
                file.write(product.code + "," + product.name + "," +
                           str(product.price) + "," + str(product.quantity) + "\n")

        self.patient_id = patient_id

class Appointment:
    def __init__(self, doctor, patient, date, time):
        self.doctor = doctor
        self.patient = patient
        self.date = date
        self.time = time


class Hospital:
    def __init__(self):
        self.appointments = []

    def book(self, doctor, patient, date, time):
        for appointment in self.appointments:
            if (appointment.doctor.doctor_id == doctor.doctor_id and
                appointment.date == date and appointment.time == time):
                print("This doctor's time slot is already booked.")
                return
        self.appointments.append(Appointment(doctor, patient, date, time))
        print("Appointment booked.")

    def cancel(self, patient_id, doctor_id, date, time):
        for appointment in self.appointments:
            if (appointment.patient.patient_id == patient_id and
                appointment.doctor.doctor_id == doctor_id and
                appointment.date == date and appointment.time == time):
                self.appointments.remove(appointment)
                print("Appointment cancelled.")
                return
        print("Appointment not found.")

    def show_doctor_appointments(self, doctor_id):
        found = False
        for appointment in self.appointments:
            if appointment.doctor.doctor_id == doctor_id:
                print(appointment.patient.name, appointment.date, appointment.time)
                found = True
        if not found:
            print("No appointments for this doctor.")

Q15. SIGNAL PROCESSING WITH NUMPY

import numpy as np
import matplotlib.pyplot as plt


def main():
    np.random.seed(10)
    time = np.linspace(0, 10, 300)
    original = np.sin(time)
    noisy = original + np.random.normal(0, 0.3, 300)

    # Moving average filter
    window = 5
    average_filter = np.ones(window) / window
    smooth = np.convolve(noisy, average_filter, mode="same")

    mean = np.mean(noisy)
    standard_deviation = np.std(noisy)
    z_score = abs((noisy - mean) / standard_deviation)
    outliers = z_score > 2

    processed = smooth.copy()
    positions = np.arange(300)
    processed[outliers] = np.interp(positions[outliers], positions[~outliers], smooth[~outliers])

    fig, ax = plt.subplots()
    ax.plot(time, noisy, label="Noisy signal")
    ax.plot(time, processed, label="Processed signal")
    ax.set_title("Signal Processing")
    ax.set_xlabel("Time")
    ax.set_ylabel("Value")
    ax.legend()

Q18. PLOT CLASS FRAMEWORK
--------------------------------------------------------------------------------
import matplotlib.pyplot as plt


class BasePlot:
    def __init__(self, x, y, title, xlabel, ylabel):
        self.x = x
        self.y = y
        self.title = title
        self.xlabel = xlabel
        self.ylabel = ylabel


class LinePlot(BasePlot):
    def render(self, ax):
        ax.plot(self.x, self.y, marker="o")
        ax.set_title(self.title)
        ax.set_xlabel(self.xlabel)
        ax.set_ylabel(self.ylabel)


class BarPlot(BasePlot):
    def render(self, ax):
        ax.bar(self.x, self.y)
        ax.set_title(self.title)
        ax.set_xlabel(self.xlabel)
        ax.set_ylabel(self.ylabel)


class Dashboard:
    def __init__(self, plots):
        self.plots = plots

    def show(self):
        fig, axes = plt.subplots(1, len(self.plots), figsize=(10, 4))
        if len(self.plots) == 1:
            axes = [axes]
        for i in range(len(self.plots)):
            self.plots[i].render(axes[i])
        plt.tight_layout()
        plt.savefig("plot_dashboard.png")
        plt.show()


line = LinePlot([1, 2, 3], [4, 6, 5], "Line Plot", "X", "Y")
bar = BarPlot(["A", "B", "C"], [5, 8, 6], "Bar Plot", "Items", "Values")
board = Dashboard([line, bar])
board.show()

import matplotlib.pyplot as plt


class Student:
    def __init__(self, name, marks):
        self.name = name
        self.marks = marks

    def average(self):
        return sum(self.marks) / 3


class ClassAnalyzer:
    def __init__(self, students):
        self.students = students

    def analyse(self):
        averages = []
        for student in self.students:
            averages.append(student.average())
        topper_index = averages.index(max(averages))
        print("Topper:", self.students[topper_index].name)
        print("Class average:", sum(averages) / len(averages))

    def draw_graphs(self):
        sorted_students = sorted(self.students, key=lambda s: s.average(), reverse=True)
        names = [s.name for s in sorted_students]
        averages = [s.average() for s in sorted_students]
        subject1 = [s.marks[0] for s in self.students]
        subject2 = [s.marks[1] for s in self.students]
        subject3 = [s.marks[2] for s in self.students]

        fig, ax = plt.subplots(1, 2, figsize=(10, 5))
        ax[0].barh(names, averages)
        ax[0].invert_yaxis()
        ax[0].set_title("Student Average Marks")
        ax[0].set_xlabel("Average")
        ax[0].set_ylabel("Student")

        ax[1].boxplot([subject1, subject2, subject3])
        ax[1].set_xticklabels(["Subject 1", "Subject 2", "Subject 3"])
        ax[1].set_title("Marks Distribution")
        ax[1].set_ylabel("Marks")

        plt.tight_layout()
        plt.savefig("student_analysis.png")
        plt.show()
        print("Observation: The first student in the sorted graph has the highest average.")
        print("The boxplot compares the spread of marks in the three subjects.")

Q20. NUMERICAL INTEGRATION AND ERROR ANALYSIS
--------------------------------------------------------------------------------
import numpy as np
import matplotlib.pyplot as plt


class Integrator:
    def __init__(self, function, a, b):
        self.function = function
        self.a = a
        self.b = b


class Trapezoid(Integrator):
    def integrate(self, n):
        x = np.linspace(self.a, self.b, n + 1)
        y = self.function(x)
        h = (self.b - self.a) / n
        return h / 2 * (y[0] + 2 * sum(y[1:-1]) + y[-1])


class Midpoint(Integrator):
    def integrate(self, n):
        h = (self.b - self.a) / n
        x = self.a + (np.arange(n) + 0.5) * h
        return h * sum(self.function(x))


def main():
    function = lambda x: np.sin(x)
    a = 0
    b = np.pi
    exact_answer = 2
    values = [4, 8, 16, 32]
    trapezoid_errors = []
    midpoint_errors = []

    for n in values:
        trap = Trapezoid(function, a, b).integrate(n)
        mid = Midpoint(function, a, b).integrate(n)
        trapezoid_errors.append(abs(exact_answer - trap))
        midpoint_errors.append(abs(exact_answer - mid))
        print("n =", n, "Trapezoid error =", trapezoid_errors[-1],
              "Midpoint error =", midpoint_errors[-1])

    fig, ax = plt.subplots()
    ax.plot(values, trapezoid_errors, marker="o", label="Trapezoid")
    ax.plot(values, midpoint_errors, marker="s", label="Midpoint")
    ax.set_title("Integration Error")
    ax.set_xlabel("Number of subintervals")
    ax.set_ylabel("Absolute error")
    ax.set_yscale("log")
    ax.legend()
    plt.savefig("integration_error.png")
    plt.show()


if __name__ == "__main__":
    main()
